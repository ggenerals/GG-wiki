倍增值域分块

~~暴力美学~~

>[P7447 [Ynoi2007] rgxsxrs](https://www.luogu.com.cn/problem/P7447)
>
>给定一个长为 $n$ 的序列 $a$，需要实现 $m$ 次操作：
>
>`1 l r x`：表示将区间 $[l,r]$ 中所有 $>x$ 的元素减去 $x$。
>
>`2 l r`：表示询问区间 $[l,r]$ 的和，最小值，最大值。

注意到标题，于是我们对值域从小到大划分为 $2^0,2^1,2^2,2^3,...$，分出了 $log\ V$ 块，每一块开一个线段树（不是权值线段树）维护。进行修改操作时，我们枚举当前处理的块$2^i,2^{i+1})$

- $x\geq 2^{i+1}$ 显然不需要任何操作。
- $x<2^i$，则这个块中的所有数都要减去 $x$，注意到一个数最多被操作 $log\ V$ 次。
具体到操作，维护当前块的 $min$，二分出其减去 $x$ 之后会掉到哪一个块，在对应的线段树中更新信息即可吗，如果现在这个数不会掉了，打上全局减 tag 即可。
- $2^i\le x<2^{i+1}$ 这个块的最大的几个数需要减去 $x$，则维护 $max$ 依次减即可。

这样做我们需要 $log V=30$ 棵 $nlogn\approx1e7$ 线段树，显然空间会炸。进行如下两个优化：

1. 对于前一个 $log V$，我们让值域分块的底数 $P$ 不再是 $2$，通过增大 $P$ 使得线段树的棵数减小，而时间复杂度仍然是 $log$ 的。

2. 我们用了 $log V$ 棵线段树，但仅仅维护了 $n$ 个点，显然有很多靠近底层的节点是空的，而这些节点占据了大部分空间。考虑当一个节点管辖的区间 $[l,r]$ 到了一个阀值，及 $r-l+1\le B$ 时，我们暴力便利进行修改，由于每个节点最多处在一个这样的序列所分成的快上（这是显然的，0 个人分块会分出重叠的块），所以复杂度只会多一个 $log$。

于是我们通过时间复杂度增加了一点常数以及一个 $log$ 的代价，将空间复杂度降到了 $O(n)$。~~一个 merge 调了我 2h~~

```cpp
#include<bits/stdc++.h>
// #define int long long
#define ll long long
using namespace std;
const int N=5e5+5,B=10,P=14,G=8,inf=2e9;
int n,m,a[N];
int MN,MX,s,t,L[N];
ll lst;
struct node{
    int mn,mx,siz,tag;
    ll v;
}tr[G][N];
#define ls(p) (p)<<1
#define rs(p) (p)<<1|1
#define mid ((l+r)>>1)
inline int get(int x){
    return upper_bound(L,L+G,x)-L-1;
}
inline void upd(int w,int p,int v){
    tr[w][p].mx-=!!tr[w][p].siz*v,
    tr[w][p].mn-=!!tr[w][p].siz*v,
    tr[w][p].v-=(ll)tr[w][p].siz*v,tr[w][p].tag+=v;
}
node merge(node A,node B){//正确写法
    A.mx=max(A.mx,B.mx),A.mn=min(A.mn,B.mn);
    A.siz=A.siz+B.siz,A.v=A.v+B.v;
    A.tag=0;//删掉这一行就会错，
//因为 x 的 tag 已经被 push_down 掉了，但是 ls/rs 的 tag 不一定下传了，所以不能直接继承
    return A;
}
void push_down(int w,int p){
    if(!tr[w][p].tag)return ;
    upd(w,ls(p),tr[w][p].tag),upd(w,rs(p),tr[w][p].tag),tr[w][p].tag=0;
}
void bdown(int w,int l,int r,int v){
    if(!v)return;
    for(int i=l;i<=r;i++)
        if(L[w]<=a[i]&&a[i]<L[w+1])a[i]-=v;
}
void bup(int w,int p,int l,int r){
    tr[w][p].siz=tr[w][p].mx=tr[w][p].v=0,tr[w][p].mn=inf;
    for(int i=l;i<=r;i++)
        if(L[w]<=a[i]&&a[i]<L[w+1])
            tr[w][p].siz++,tr[w][p].v+=a[i],
            tr[w][p].mx=max(tr[w][p].mx,a[i]),
            tr[w][p].mn=min(tr[w][p].mn,a[i]);
}
void ins(int w,int p,int l,int r,int x,int v){
    if(r-l<B){
        bdown(w,l,r,tr[w][p].tag),tr[w][p].tag=0;
        return a[x]=v,bup(w,p,l,r);
    }
    push_down(w,p);
    if(x<=mid)ins(w,ls(p),l,mid,x,v); 
    else ins(w,rs(p),mid+1,r,x,v);
    tr[w][p]=merge(tr[w][ls(p)],tr[w][rs(p)]);
}
inline void bupd(int w,int l,int r,int v){
    for(int i=l;i<=r;i++)
        if(L[w]<=a[i]&&a[i]<L[w+1]&&a[i]>v)
            a[i]-=v,a[i]<L[w]&&(ins(get(a[i]),1,1,n,i,a[i]),0);
}
inline void bquery(int w,int l,int r){
    for(int i=l;i<=r;i++)if(L[w]<=a[i]&&a[i]<L[w+1])
        lst+=a[i],MN=min(MN,a[i]),MX=max(MX,a[i]);
}
void update(int w,int p,int l,int r,int v){
    if(tr[w][p].mx<=v)return;
    if(s<=l&&r<=t&&tr[w][p].mn-v>=L[w])return upd(w,p,v);
    if(r-l<B){
        bdown(w,l,r,tr[w][p].tag),tr[w][p].tag=0;
        return bupd(w,max(l,s),min(r,t),v),bup(w,p,l,r);
    }
    push_down(w,p);
    if(s<=mid)update(w,ls(p),l,mid,v);
    if(t>mid)update(w,rs(p),mid+1,r,v);
    tr[w][p]=merge(tr[w][ls(p)],tr[w][rs(p)]);
}
void query(int w,int p,int l,int r){
    if(!tr[w][p].siz)return;
    if(s<=l&&r<=t)
        return lst+=tr[w][p].v,MX=max(MX,tr[w][p].mx),MN=min(MN,tr[w][p].mn),void();
    if(r-l<B){
        bdown(w,l,r,tr[w][p].tag),tr[w][p].tag=0;
        bup(w,p,l,r); 
        return bquery(w,max(l,s),min(r,t));
    }
    push_down(w,p);
    if(s<=mid)query(w,ls(p),l,mid);
    if(t>mid)query(w,rs(p),mid+1,r);
}
void build(int w,int p,int l,int r){
    if(r-l<B)return bup(w,p,l,r);
    build(w,ls(p),l,mid),build(w,rs(p),mid+1,r);
    tr[w][p]=merge(tr[w][ls(p)],tr[w][rs(p)]);
}
signed main(){
    ios::sync_with_stdio(0);cin.tie(0);
	cin>>n>>m,L[0]=1;
	for(int i=1;i<=n;i++)cin>>a[i];
	for(int i=1;i<=G;i++)L[i]=L[i-1]*P;
	for(int w=0;w<G;w++)build(w,1,1,n);
    ll ss=0;
	while(m--){
		int op,x;cin>>op>>s>>t;
        s^=ss,t^=ss;
		if(op&1){
			cin>>x;
            x^=ss;
			for(int w=0;w<G;w++)update(w,1,1,n,x);
		}
		else{
			lst=MX=0,MN=1e9;
			for(int w=0;w<G;w++)query(w,1,1,n);
			cout<<(ss=lst)<<" "<<MN<<" "<<MX<<"\n",ss%=(1ll<<20);
		}
	}
    return 0;
}
```
